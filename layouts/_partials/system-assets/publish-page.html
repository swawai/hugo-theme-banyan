{{- $ctx := .ctx -}} {{- $src := trim (printf "%v" (.src | default "")) " \r\n\t" -}} {{- $required := true -}} {{- if isset . "required" -}}{{- $required = .required -}}{{- end -}} {{- $result := dict "resolved" false "src" $src -}} {{- $isExternal := or (hasPrefix $src "//") (ne (replaceRE `^[A-Za-z][A-Za-z0-9+.-]*:.*$` "" $src) $src) -}} {{- $isRootRelative := hasPrefix $src "/" -}} {{- if and (ne $src "") (not $isExternal) (not $isRootRelative) -}} {{- $resourcePath := replaceRE `[#?].*$` "" $src -}} {{- $resourcePath = strings.TrimPrefix "./" $resourcePath -}} {{- $page := $ctx.Page -}} {{- $resource := "" -}} {{- with $page.Resources -}} {{- $resource = .Get $resourcePath -}} {{- end -}} {{- if not $resource -}} {{- if $required -}} {{- errorf "Build failed: page %q cannot resolve local page resource %q. Put the file in the same page bundle, or use an absolute/remote URL." ($page.File.Path | default $page.RelPermalink) $src -}} {{- end -}} {{- else -}} {{- $pathInfo := partial "system-assets/page-output-path.html" $page -}} {{- $targetDir := index $pathInfo "targetDir" -}} {{- $targetPath := path.Join $targetDir $resource.Name -}} {{- $published := $resource | resources.Copy $targetPath | fingerprint -}} {{- $result = dict "resolved" true "resource" $resource "src" $published.RelPermalink "targetDir" $targetDir -}} {{- end -}} {{- end -}} {{- return $result -}}
