{{- $ctx := .ctx -}} {{- $src := trim (printf "%v" (.src | default "")) " \r\n\t" -}} {{- $result := dict -}} {{- $isExternal := or (hasPrefix $src "//") (ne (replaceRE `^[A-Za-z][A-Za-z0-9+.-]*:.*$` "" $src) $src) -}} {{- if eq $src "" -}} {{- errorf "Build failed: asset publisher requires src." -}} {{- else if or $isExternal (hasPrefix $src "/") -}} {{- errorf "Build failed: asset src %q must be a local page bundle resource or a Hugo asset path; static and remote URLs are not supported." $src -}} {{- else if in $src "\\" -}} {{- errorf "Build failed: asset src %q must use forward slashes." $src -}} {{- else -}} {{- $resourcePath := replaceRE `[#?].*$` "" $src -}} {{- $explicitPageResource := hasPrefix $resourcePath "./" -}} {{- $resourcePath = strings.TrimPrefix "./" $resourcePath -}} {{- if or (eq $resourcePath "") (in (split $resourcePath "/") "..") -}} {{- errorf "Build failed: asset src %q must not be empty or contain parent path segments." $src -}} {{- else -}} {{- /* errorf logs a build error; it does not stop subsequent template actions. */ -}} {{- $page := $ctx.Page -}} {{- $pagePublished := partial "system-assets/publish-page.html" (dict "ctx" $ctx "src" $src "required" false) -}} {{- if index $pagePublished "resolved" -}} {{- $result = $pagePublished -}} {{- else if $explicitPageResource -}} {{- errorf "Build failed: page %q cannot resolve page bundle asset %q." ($page.File.Path | default $page.RelPermalink) $src -}} {{- else -}} {{- with resources.Get $resourcePath -}} {{- $published := . | fingerprint -}} {{- $result = dict "resource" . "published" $published "src" $published.RelPermalink "source" "assets" -}} {{- else -}} {{- errorf "Build failed: asset src %q was not found in page bundle %q or Hugo assets." $src ($page.File.Path | default $page.RelPermalink) -}} {{- end -}} {{- end -}} {{- end -}} {{- end -}} {{- return $result -}}
