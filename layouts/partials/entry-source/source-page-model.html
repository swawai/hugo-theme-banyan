) -}} {{- end -}} {{- $levelModel := dict "item" $item -}} {{- if gt (len $collectionSource) 0 -}} {{- $levelModel = merge $levelModel (dict "collection_source" $collectionSource) -}} {{- end -}} {{- $levels = $levels | append $levelModel -}} {{- $itemHref := trim (printf "%v" (index $item "href" | default "")) " " -}} {{- $itemText := trim (printf "%v" (index $item "text" | default "")) " " -}} dict -}} {{- $entryHrefs := slice -}} {{- if in (slice "all" "products" "section-d" "taxonomy") $provider -}} {{- $currentCollectionItemsJson := partial (printf "entry-source/items-json-%s.html" $provider) (dict "source" (dict "page" $page "provider" $provider "logical_path" $logicalPath ) ) -}} {{- if ne (trim $currentCollectionItemsJson " \r\n\t") "" -}} {{- $currentCollectionItems = slice -}} {{- end -}} {{- end -}} {{- $result := dict "provider" $provider "page" $page "logical_path" $logicalPath "root_item" $rootItem "tail_items" $tailItems "levels" $levels
