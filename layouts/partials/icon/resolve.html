{{- /* Validate a named SVG, text, or image value. Page resource paths are resolved by icon/page-value.html. */ -}} {{- $value := .value -}} {{- $context := .context | default "icon" -}} {{- $resolved := "" -}} {{- if reflect.IsMap $value -}} {{- if or (ne (len $value) 1) (not (or (isset $value "text") (isset $value "image"))) -}} {{- errorf "Build failed: %s must be an SVG name or an object containing only text or image." $context -}} {{- else -}} {{- range $field, $raw := $value -}} {{- if ne (printf "%T" $raw) "string" -}} {{- errorf "Build failed: %s.%s must be a non-empty string." $context $field -}} {{- else -}} {{- $text := trim $raw " \r\n\t" -}} {{- if not $text -}}{{- errorf "Build failed: %s.%s must be a non-empty string." $context $field -}}{{- end -}} {{- $resolved = dict $field $text -}} {{- end -}} {{- end -}} {{- end -}} {{- else if eq (printf "%T" $value) "string" -}} {{- $resolved = lower (trim $value " \r\n\t") -}} {{- if and $resolved (not (index site.Data.icons $resolved)) -}} {{- errorf "Build failed: %s declares undefined icon %q; define it in data/icons.toml." $context $resolved -}} {{- end -}} {{- else -}} {{- errorf "Build failed: %s must be an SVG name or an object containing only text or image." $context -}} {{- end -}} {{- return $resolved -}}
