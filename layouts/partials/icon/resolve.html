{{- /* Validate a named SVG, text, or image value. Page resource paths are resolved by icon/page-value.html. */ -}} {{- $value := .value -}} {{- $context := .context | default "icon" -}} {{- $resolved := "" -}} {{- if reflect.IsMap $value -}} {{- $isText := and (isset $value "text") (eq (len $value) 1) -}} {{- $isImage := and (isset $value "image") (or (eq (len $value) 1) (and (eq (len $value) 2) (isset $value "monochrome"))) -}} {{- if not (or $isText $isImage) -}} {{- errorf "Build failed: %s must be an SVG name or an object containing only text, or image with optional monochrome." $context -}} {{- else -}} {{- $field := cond $isImage "image" "text" -}} {{- $raw := index $value $field -}} {{- if ne (printf "%T" $raw) "string" -}} {{- errorf "Build failed: %s.%s must be a non-empty string." $context $field -}} {{- else -}} {{- $text := trim $raw " \r\n\t" -}} {{- if not $text -}}{{- errorf "Build failed: %s.%s must be a non-empty string." $context $field -}}{{- end -}} {{- $resolved = dict $field $text -}} {{- end -}} {{- if isset $value "monochrome" -}} {{- if ne (printf "%T" $value.monochrome) "bool" -}} {{- errorf "Build failed: %s.monochrome must be a boolean." $context -}} {{- else if and $value.monochrome (reflect.IsMap $resolved) -}} {{- $resolved = merge $resolved (dict "monochrome" true) -}} {{- end -}} {{- end -}} {{- end -}} {{- else if eq (printf "%T" $value) "string" -}} {{- $resolved = lower (trim $value " \r\n\t") -}} {{- if and $resolved (not (index site.Data.icons $resolved)) -}} {{- errorf "Build failed: %s declares undefined icon %q; define it in data/icons.toml." $context $resolved -}} {{- end -}} {{- else -}} {{- errorf "Build failed: %s must be an SVG name or an object containing only text, or image with optional monochrome." $context -}} {{- end -}} {{- return $resolved -}}
