{{- $ctx := .ctx | default . -}} {{- /* Navigation comes from real pages; slots enable path and article metadata. */ -}} {{- $slotConfig := dict -}} {{- with index $ctx.Params "slots" -}} {{- if not (reflect.IsMap .) -}} {{- errorf "Build failed: page %q must define slots as a map/object." ($ctx.File.Path | default (printf "%s %s" $ctx.Kind $ctx.RelPermalink)) -}} {{- end -}} {{- $slotConfig = . -}} {{- end -}} {{- $supportedSlots := slice "breadcrumb" "meta" -}} {{- $supportedLookup := dict -}} {{- range $slot := $supportedSlots -}} {{- $supportedLookup = merge $supportedLookup (dict $slot true) -}} {{- end -}} {{- range $slotName, $_ := $slotConfig -}} {{- $key := printf "%v" $slotName -}} {{- if not (isset $supportedLookup $key) -}} {{- errorf "Build failed: page %q does not support slots.%s." ($ctx.File.Path | default (printf "%s %s" $ctx.Kind $ctx.RelPermalink)) $key -}} {{- end -}} {{- end -}} {{- $slotFlags := dict -}} {{- range $slot := $supportedSlots -}} {{- $slotFlags = merge $slotFlags (dict $slot false) -}} {{- if not (isset $slotConfig $slot) -}}{{- continue -}}{{- end -}} {{- $raw := index $slotConfig $slot -}} {{- if ne (printf "%T" $raw) "bool" -}} {{- errorf "Build failed: page %q only supports true/false for slots.%s." ($ctx.File.Path | default (printf "%s %s" $ctx.Kind $ctx.RelPermalink)) $slot -}} {{- else -}} {{- $slotFlags = merge $slotFlags (dict $slot $raw) -}} {{- end -}} {{- end -}} {{- $ctx.Scratch.Set "slot_flags" $slotFlags -}} {{- return $slotFlags -}}
